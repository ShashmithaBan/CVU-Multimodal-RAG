"""Debug script for OCR extraction."""
import json

with open('outputs/OPaLnMw2i_0_chunks_metadata.json') as f:
    data = json.load(f)
chunks = data['chunks']

print(f"Total chunks: {len(chunks)}")

# Check Projects component
target = 'projects'
code_frames = []
all_code_lines = set()

for i, chunk in enumerate(chunks):
    ocr = chunk.get('ocr_text', '')
    ocr_lower = ocr.lower()
    
    if target in ocr_lower:
        has_code = any(kw in ocr_lower for kw in ['import ', 'const ', 'function ', 'export ', 'return '])
        if has_code and len(ocr) > 200:
            code_frames.append(chunk)
            # Extract code lines
            for line in ocr.split('\n'):
                line_clean = line.strip()
                if line_clean and len(line_clean) > 5:
                    skip_ui = any(x in line_clean for x in ['localhost', 'File Edit', 'Visual Studio', 'EXPL', 'PROBLEMS'])
                    if not skip_ui:
                        all_code_lines.add(line_clean)

print(f"Code frames found: {len(code_frames)}")

if code_frames:
    print("\n--- First code frame OCR (raw repr) ---")
    sample = code_frames[0].get('ocr_text', '')
    print(f"Newlines in OCR: {sample.count(chr(10))}")
    print(f"Length: {len(sample)}")
    print(repr(sample[:500]))
