# Docker Compose for Multimodal Video Retrieval System
# Usage: docker-compose up --build

version: '3.8'

services:
  # Backend API Service
  backend:
    build:
      context: ./backend
      dockerfile: Dockerfile
    container_name: videoai-backend
    ports:
      - "5000:5000"
    volumes:
      # Persist downloads and outputs
      - ./backend/downloads:/app/downloads
      - ./backend/outputs:/app/outputs
    environment:
      - FLASK_ENV=production
      - PYTHONUNBUFFERED=1
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:5000/api/library/videos"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 120s
    deploy:
      resources:
        limits:
          memory: 8G
        reservations:
          memory: 4G

  # Frontend React App
  frontend:
    build:
      context: ./my-app
      dockerfile: Dockerfile
    container_name: videoai-frontend
    ports:
      - "3000:80"
    depends_on:
      - backend
    restart: unless-stopped

networks:
  default:
    name: videoai-network

volumes:
  downloads:
  outputs:
